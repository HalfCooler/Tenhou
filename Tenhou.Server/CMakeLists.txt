cmake_minimum_required(VERSION 3.24)
project(Tenhou.Server VERSION 0.27.0 LANGUAGES CXX)
if (NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows and MSVC are required.")
endif ()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(authority_rules STATIC common.cpp scoring.cpp game.cpp)
target_compile_options(authority_rules PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(authority_rules PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(tenhou_server main.cpp storage.cpp service.cpp accounts.cpp)
target_compile_options(tenhou_server PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(tenhou_server PRIVATE authority_rules ws2_32 bcrypt crypt32 advapi32)
configure_file(server.ini server.ini COPYONLY)
set(server_destination .)
if (NOT CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(server_destination server)
endif ()
install(TARGETS tenhou_server RUNTIME DESTINATION ${server_destination})
install(FILES server.ini README.md start-server.cmd stop-server.cmd manage_native_server.ps1 DESTINATION ${server_destination})
